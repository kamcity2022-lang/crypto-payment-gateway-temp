using System;

namespace Gateway.Domain.Entities;

public class BlockchainScanCursor
{
    public string Network { get; private set; } = string.Empty;
    public string Address { get; private set; } = string.Empty;
    public long LastProcessedBlockTimestamp { get; private set; }
    public ulong LastProcessedBlockNumber { get; private set; }
    public string? LastFingerprint { get; private set; }
    public DateTime UpdatedAtUtc { get; private set; }

    private BlockchainScanCursor() { }

    public BlockchainScanCursor(string network, string address, long initialTimestamp, ulong initialBlockNumber)
    {
        Network = network;
        Address = address;
        LastProcessedBlockTimestamp = initialTimestamp;
        LastProcessedBlockNumber = initialBlockNumber;
        LastFingerprint = null;
        UpdatedAtUtc = DateTime.UtcNow;
    }

    public void UpdateCursor(long blockTimestamp, ulong blockNumber)
    {
        if (blockTimestamp >= LastProcessedBlockTimestamp)
        {
            LastProcessedBlockTimestamp = blockTimestamp;
            LastProcessedBlockNumber = Math.Max(LastProcessedBlockNumber, blockNumber);
            UpdatedAtUtc = DateTime.UtcNow;
        }
    }

    public void SetFingerprint(string? fingerprint)
    {
        LastFingerprint = fingerprint;
        UpdatedAtUtc = DateTime.UtcNow;
    }
}

using System;
using System.Collections.Generic;
using System.Linq;
using Gateway.Domain.Enums;

namespace Gateway.Domain.Entities;

public class Invoice
{
    public Guid Id { get; private set; }
    public Guid MerchantId { get; private set; }
    public Guid DerivedAddressId { get; private set; }
    public string DepositAddress { get; private set; } = string.Empty;
    public string Currency { get; private set; } = "USDT";
    public string Network { get; private set; } = "TRON";

    public decimal FiatAmount { get; private set; }
    public string FiatCurrency { get; private set; } = "USD";
    public decimal ExchangeRate { get; private set; }
    public string ExchangeRateSource { get; private set; } = string.Empty;
    public DateTime ExchangeRateTimestampUtc { get; private set; }

    public long ExpectedAmountUnits { get; private set; }
    public long PaidAmountUnits { get; private set; }
    public long UnderpaymentThresholdUnits { get; private set; }

    public InvoiceStatus Status { get; private set; }
    public string CallbackUrl { get; private set; } = string.Empty;
    public string CheckoutToken { get; private set; } = string.Empty;

    public DateTime ExpiresAtUtc { get; private set; }
    public DateTime CreatedAtUtc { get; private set; }
    public DateTime? PaidAtUtc { get; private set; }

    private readonly List<PaymentTransfer> _transfers = new();
    public IReadOnlyCollection<PaymentTransfer> Transfers => _transfers.AsReadOnly();

    private Invoice() { }

    public Invoice(
        Guid merchantId,
        Guid derivedAddressId,
        string depositAddress,
        decimal fiatAmount,
        string fiatCurrency,
        decimal exchangeRate,
        string exchangeRateSource,
        DateTime exchangeRateTimestampUtc,
        long expectedAmountUnits,
        long underpaymentThresholdUnits,
        string callbackUrl,
        string checkoutToken,
        TimeSpan validityDuration)
    {
        if (fiatAmount <= 0)
            throw new ArgumentOutOfRangeException(nameof(fiatAmount), "Fiat amount must be positive.");
        if (expectedAmountUnits <= 0)
            throw new ArgumentOutOfRangeException(nameof(expectedAmountUnits), "Expected crypto units must be positive.");

        Id = Guid.NewGuid();
        MerchantId = merchantId;
        DerivedAddressId = derivedAddressId;
        DepositAddress = depositAddress;
        FiatAmount = fiatAmount;
        FiatCurrency = fiatCurrency;
        ExchangeRate = exchangeRate;
        ExchangeRateSource = exchangeRateSource;
        ExchangeRateTimestampUtc = exchangeRateTimestampUtc;
        ExpectedAmountUnits = expectedAmountUnits;
        UnderpaymentThresholdUnits = underpaymentThresholdUnits;
        PaidAmountUnits = 0;
        Status = InvoiceStatus.Pending;
        CallbackUrl = callbackUrl;
        CheckoutToken = checkoutToken;
        CreatedAtUtc = DateTime.UtcNow;
        ExpiresAtUtc = CreatedAtUtc.Add(validityDuration);
    }

    public bool RegisterTransfer(PaymentTransfer transfer)
    {
        if (transfer == null) throw new ArgumentNullException(nameof(transfer));

        // Idempotency: Ignore duplicate transfers matching (Network, TxHash, TransferIndex)
        if (_transfers.Any(t => t.Network == transfer.Network &&
                                t.TransactionHash.Equals(transfer.TransactionHash, StringComparison.OrdinalIgnoreCase) &&
                                t.TransferIndex == transfer.TransferIndex))
        {
            return false;
        }

        _transfers.Add(transfer);
        EvaluateStatus();
        return true;
    }

    public InvoiceStatus EvaluateStatus()
    {
        var confirmedUnits = _transfers
            .Where(t => t.Status == PaymentStatus.Confirmed)
            .Sum(t => t.AmountUnits);

        PaidAmountUnits = confirmedUnits;
        var now = DateTime.UtcNow;

        // If invoice is already completed/cancelled, do not revert
        if (Status == InvoiceStatus.Cancelled)
            return Status;

        if (PaidAmountUnits == 0)
        {
            if (now > ExpiresAtUtc && Status == InvoiceStatus.Pending)
            {
                Status = InvoiceStatus.Expired;
            }
            return Status;
        }

        var effectiveExpected = ExpectedAmountUnits - UnderpaymentThresholdUnits;

        if (PaidAmountUnits >= ExpectedAmountUnits)
        {
            Status = PaidAmountUnits > ExpectedAmountUnits ? InvoiceStatus.Overpaid : InvoiceStatus.Paid;
            PaidAtUtc ??= DateTime.UtcNow;
        }
        else if (PaidAmountUnits >= effectiveExpected)
        {
            Status = InvoiceStatus.Paid;
            PaidAtUtc ??= DateTime.UtcNow;
        }
        else
        {
            Status = now > ExpiresAtUtc ? InvoiceStatus.Underpaid : InvoiceStatus.PartiallyPaid;
        }

        return Status;
    }

    public void MarkAsExpiredIfPastDue()
    {
        if (Status == InvoiceStatus.Pending && DateTime.UtcNow > ExpiresAtUtc && PaidAmountUnits == 0)
        {
            Status = InvoiceStatus.Expired;
        }
    }
}

using System;

namespace Gateway.Domain.Entities;

public class OutboxEvent
{
    public Guid Id { get; private set; }
    public Guid AggregateId { get; private set; }
    public string EventType { get; private set; } = string.Empty;
    public string Payload { get; private set; } = string.Empty;
    public string DestinationUrl { get; private set; } = string.Empty;
    public string SecretKey { get; private set; } = string.Empty;
    public int AttemptCount { get; private set; }
    public bool IsProcessed { get; private set; }
    public DateTime CreatedAtUtc { get; private set; }
    public DateTime? NextAttemptAtUtc { get; private set; }
    public string? LastError { get; private set; }

    private OutboxEvent() { }

    public OutboxEvent(Guid aggregateId, string eventType, string payload, string destinationUrl, string secretKey)
    {
        Id = Guid.NewGuid();
        AggregateId = aggregateId;
        EventType = eventType;
        Payload = payload;
        DestinationUrl = destinationUrl;
        SecretKey = secretKey;
        AttemptCount = 0;
        IsProcessed = false;
        CreatedAtUtc = DateTime.UtcNow;
        NextAttemptAtUtc = CreatedAtUtc;
    }

    public void RecordFailure(string error, int maxRetries = 5)
    {
        AttemptCount++;
        LastError = error;

        if (AttemptCount >= maxRetries)
        {
            // Abandon after reaching maximum attempts
            NextAttemptAtUtc = null;
        }
        else
        {
            // Exponential backoff: 10s, 40s, 90s, 160s...
            var delaySeconds = Math.Pow(AttemptCount, 2) * 10;
            NextAttemptAtUtc = DateTime.UtcNow.AddSeconds(delaySeconds);
        }
    }

    public void RecordSuccess()
    {
        IsProcessed = true;
        NextAttemptAtUtc = null;
        LastError = null;
    }
}

using System;
using Gateway.Domain.Enums;

namespace Gateway.Domain.Entities;

public class PaymentTransfer

    public Guid Id { get; private set; }
    public Guid InvoiceId { get; private set; }
    public string Network { get; private set; } = string.Empty;
    public string TransactionHash { get; private set; } = string.Empty;
    public int TransferIndex { get; private set; }
    public string TokenContractAddress { get; private set; } = string.Empty;
    public string FromAddress { get; private set; } = string.Empty;
    public string ToAddress { get; private set; } = string.Empty;
    public long AmountUnits { get; private set; } // مقدار دقیق آن‌چین (واحد Sun: 1 USDT = 1,000,000 Sun)
    public ulong BlockNumber { get; private set; }
    public ulong Confirmations { get; private set; }
    public PaymentStatus Status { get; private set; }
    public DateTime ObservedAtUtc { get; private set; }
    public DateTime? ConfirmedAtUtc { get; private set; }

    private PaymentTransfer() { }

    public PaymentTransfer(
        Guid invoiceId,
        string network,
        string transactionHash,
        int transferIndex,
        string tokenContractAddress,
        string fromAddress,
        string toAddress,
        long amountUnits,
        ulong blockNumber)
    {
        Id = Guid.NewGuid();
        InvoiceId = invoiceId;
        Network = network;
        TransactionHash = transactionHash;
        TransferIndex = transferIndex;
        TokenContractAddress = tokenContractAddress;
        FromAddress = fromAddress;
        ToAddress = toAddress;
        AmountUnits = amountUnits;
        BlockNumber = blockNumber;
        Status = PaymentStatus.Observed;
        ObservedAtUtc = DateTime.UtcNow;
    }

    public void UpdateConfirmations(ulong currentBlockNumber, ulong requiredConfirmations)
    {
        if (currentBlockNumber >= BlockNumber)
        {
            Confirmations = (currentBlockNumber - BlockNumber) + 1;
            if (Confirmations >= requiredConfirmations && Status != PaymentStatus.Confirmed)
            {
                Status = PaymentStatus.Confirmed;
                ConfirmedAtUtc = DateTime.UtcNow;
            }
            else if (Status == PaymentStatus.Observed)
            {
                Status = PaymentStatus.Confirming;
            }
        }
    }

    public void MarkAsFailed()
    {
        Status = PaymentStatus.Failed;
    }

    public void MarkAsIgnored()
    {
        Status = PaymentStatus.Ignored;
    }
}

using System;
using System.Security.Cryptography;
using NBitcoin;

namespace Gateway.Blockchain.Tron;

public class TronAddressDerivationService
{
    private readonly ExtPubKey _extPubKey;

    public TronAddressDerivationService(string accountXPub)
    {
        if (string.IsNullOrWhiteSpace(accountXPub))
            throw new ArgumentException("Account xPub cannot be empty.", nameof(accountXPub));

        var btcExtPubKey = new BitcoinExtPubKey(accountXPub.Trim(), Network.Main);
        _extPubKey = btcExtPubKey.ExtPubKey;
    }

    public string DeriveAddress(uint index)
    {
        // اشتقاق آدرس عمومی فرزند در شاخه: 0 / index
        var childPubKey = _extPubKey.Derive(0).Derive(index).PubKey;
        return PublicKeyToTronAddress(childPubKey);
    }

    public static string PublicKeyToTronAddress(PubKey pubKey)
    {
        // استخراج ۶۴ بایت کلید عمومی خام (حذف بایت 0x04)
        var uncompressed = pubKey.Decompress().ToBytes();
        var rawKey = new byte[64];
        Buffer.BlockCopy(uncompressed, 1, rawKey, 0, 64);

        // محاسبه هش استاندارد Keccak-256
        var hash = Keccak256(rawKey);

        // ۲۰ بایت انتهای هش به همراه پیشوند 0x41 شبکه ترون (مجموعاً ۲۱ بایت)
        var addressPayload = new byte[21];
        addressPayload[0] = 0x41;
        Buffer.BlockCopy(hash, 12, addressPayload, 1, 20);

        // انکود Base58Check به فرمت آدرس استاندارد ترون (شروع با T)
        return Base58CheckEncode(addressPayload);
    }

    private static string Base58CheckEncode(byte[] payload)
    {
        using var sha256 = SHA256.Create();
        var hash1 = sha256.ComputeHash(payload);
        var hash2 = sha256.ComputeHash(hash1);

        var dataWithChecksum = new byte[payload.Length + 4];
        Buffer.BlockCopy(payload, 0, dataWithChecksum, 0, payload.Length);
        Buffer.BlockCopy(hash2, 0, dataWithChecksum, payload.Length, 4);

        return ToBase58(dataWithChecksum);
    }

    private static string ToBase58(byte[] data)
    {
        const string alphabet = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
        var intData = new System.Numerics.BigInteger(data, isUnsigned: true, isBigEndian: true);
        var result = string.Empty;

        while (intData > 0)
        {
            intData = System.Numerics.BigInteger.DivRem(intData, 58, out var remainder);
            result = alphabet[(int)remainder] + result;
        }

        for (var i = 0; i < data.Length && data[i] == 0; i++)
        {
            result = '1' + result;
        }

        return result;
    }

    #region Pure Keccak-256 Implementation
    private static readonly ulong[] RoundConstants =
    [
        0x0000000000000001UL, 0x0000000000008082UL, 0x800000000000808aUL, 0x8000000080008000UL,
        0x000000000000808bUL, 0x0000000080000001UL, 0x8000000080008081UL, 0x8000000000008009UL,
        0x000000000000008aUL, 0x0000000000000088UL, 0x0000000080008009UL, 0x000000008000000aUL,
        0x000000008000808bUL, 0x800000000000008bUL, 0x8000000000008089UL, 0x8000000000008003UL,
        0x8000000000008002UL, 0x8000000000000080UL, 0x000000000000800aUL, 0x800000008000000aUL,
        0x8000000080008081UL, 0x8000000000008080UL, 0x0000000080000001UL, 0x8000000080008008UL
    ];

    private static readonly int[] RhoOffsets =
    [
        0, 1, 62, 28, 27, 36, 44, 6, 55, 20, 3, 10, 43, 25, 39, 41, 45, 15, 21, 8, 18, 2, 61, 56, 14
    ];

    private static byte[] Keccak256(byte[] message)
    {
        const int rate = 136; // 1088 bits
        var state = new ulong[25];
        var padded = Pad(message, rate);

        for (var i = 0; i < padded.Length; i += rate)
        {
            for (var j = 0; j < rate / 8; j++)
            {
                state[j] ^= BitConverter.ToUInt64(padded, i + j * 8);
            }
            KeccakF1600(state);
        }

        var output = new byte[32];
        for (var i = 0; i < 4; i++)
        {
            var bytes = BitConverter.GetBytes(state[i]);
            Buffer.BlockCopy(bytes, 0, output, i * 8, 8);
        }

        return output;
    }

    private static byte[] Pad(byte[] data, int rate)
    {
        var padLen = rate - (data.Length % rate);
        if (padLen == 0) padLen = rate;

        var result = new byte[data.Length + padLen];
        Buffer.BlockCopy(data, 0, result, 0, data.Length);

        if (padLen == 1)
        {
            result[data.Length] = 0x81;
        }
        else
        {
            result[data.Length] = 0x01; // استاندارد Keccak-256 (EVM/TRON)
            result[^1] = 0x80;
        }

        return result;
    }

    private static void KeccakF1600(ulong[] a)
    {
        var c = new ulong[5];
        var d = new ulong[5];
        var b = new ulong[25];

        for (var round = 0; round < 24; round++)
        {
            for (var x = 0; x < 5; x++)
                c[x] = a[x] ^ a[x + 5] ^ a[x + 10] ^ a[x + 15] ^ a[x + 20];

            for (var x = 0; x < 5; x++)
                d[x] = c[(x + 4) % 5] ^ RotL(c[(x + 1) % 5], 1);

            for (var i = 0; i < 25; i++)
                a[i] ^= d[i % 5];

            for (var i = 0; i < 25; i++)
                b[PiIndex(i)] = RotL(a[i], RhoOffsets[i]);

            for (var y = 0; y < 25; y += 5)
            {
                for (var x = 0; x < 5; x++)
                    a[y + x] = b[y + x] ^ (~b[y + (x + 1) % 5] & b[y + (x + 2) % 5]);
            }

            a[0] ^= RoundConstants[round];
        }
    }

    private static int PiIndex(int index)
    {
        var x = index % 5;
        var y = index / 5;
        return ((0 * x + 1 * y) % 5) + 5 * ((2 * x + 3 * y) % 5);
    }

    private static ulong RotL(ulong val, int shift) => (val << shift) | (val >> (64 - shift));
    #endregion
}

    private async Task ScanInvoiceWithPaginationAsync(Invoice invoice, ulong solidifiedBlockNumber, CancellationToken ct)
    {
        var cursor = await _dbContext.BlockchainScanCursors
            .FirstOrDefaultAsync(c => c.Network == "TRON" && c.Address == invoice.DepositAddress, ct);

        if (cursor == null)
        {
            cursor = new BlockchainScanCursor("TRON", invoice.DepositAddress, 0, 0);
            _dbContext.BlockchainScanCursors.Add(cursor);
        }

        string? fingerprint = cursor.LastFingerprint;
        bool hasMorePages = true;

        while (hasMorePages)
        {
            var url = $"https://api.trongrid.io/v1/accounts/{invoice.DepositAddress}/transactions/trc20?contract_address={_usdtContractAddress}&limit=20";
            if (!string.IsNullOrEmpty(fingerprint))
            {
                url += $"&fingerprint={Uri.EscapeDataString(fingerprint)}";
            }
            else if (cursor.LastProcessedBlockTimestamp > 0)
            {
                url += $"&min_timestamp={cursor.LastProcessedBlockTimestamp}";
            }

            using var request = new HttpRequestMessage(HttpMethod.Get, url);
            using var response = await _httpClient.SendAsync(request, ct);

            if (!response.IsSuccessStatusCode)
            {
                _logger.LogWarning("TronGrid query returned {Status} for address {Address}", response.StatusCode, invoice.DepositAddress);
                return; // Stop pagination on error; do not advance cursor
            }

            var json = await response.Content.ReadAsStringAsync(ct);
            using var doc = JsonDocument.Parse(json);

            if (!doc.RootElement.TryGetProperty("data", out var dataElement) || dataElement.GetArrayLength() == 0)
            {
                break;
            }

            foreach (var item in dataElement.EnumerateArray())
            {
                var toAddress = item.GetProperty("to").GetString() ?? string.Empty;
                if (string.IsNullOrEmpty(toAddress) || !string.Equals(toAddress, invoice.DepositAddress, StringComparison.OrdinalIgnoreCase))
                    continue;

                var txHash = item.GetProperty("transaction_id").GetString() ?? string.Empty;
                if (string.IsNullOrEmpty(txHash)) continue;

                // Establish strict event/log index
                if (!item.TryGetProperty("event_index", out var idxEl) || !idxEl.TryGetInt32(out var eventIndex))
                {
                    _logger.LogError("Transfer event index could not be determined for Tx {TxHash}. Skipping to prevent duplicate accounting.", txHash);
                    continue;
                }

                var rawValueStr = item.GetProperty("value").GetString() ?? "0";
                if (!long.TryParse(rawValueStr, out var amountUnits) || amountUnits <= 0) continue;

                var blockTimestamp = item.GetProperty("block_timestamp").GetInt64();

                // Reorganization protection: Verify transaction via SolidityNode
                var txDetail = await VerifySolidifiedTransactionAsync(txHash, ct);
                if (!txDetail.IsSolidified || !txDetail.IsSuccess)
                {
                    _logger.LogWarning("Tx {TxHash} not solidified or execution failed. Skipping transfer credit.", txHash);
                    continue;
                }

                var existingTransfer = await _dbContext.PaymentTransfers
                    .FirstOrDefaultAsync(t => t.Network == "TRON" && t.TransactionHash == txHash && t.TransferIndex == eventIndex, ct);

                if (existingTransfer == null)
                {
                    var fromAddress = item.GetProperty("from").GetString() ?? string.Empty;
                    var transfer = new PaymentTransfer(
                        invoice.Id,
                        "TRON",
                        txHash,
                        eventIndex,
                        _usdtContractAddress,
                        fromAddress,
                        toAddress,
                        amountUnits,
                        txDetail.BlockNumber);

                    _dbContext.PaymentTransfers.Add(transfer);
                    var registered = invoice.RegisterTransfer(transfer);

                    if (registered)
                    {
                        transfer.UpdateConfirmations(solidifiedBlockNumber, 1); // Confirmed in solidified chain
                        var previousStatus = invoice.Status;
                        var currentStatus = invoice.EvaluateStatus();

                        if (previousStatus != currentStatus)
                        {
                            var merchant = await _dbContext.Merchants.FindAsync(new object[] { invoice.MerchantId }, ct);
                            if (merchant != null)
                            {
                                var payload = JsonSerializer.Serialize(new
                                {
                                    @event = $"invoice.{currentStatus.ToString().ToLowerInvariant()}",
                                    invoiceId = invoice.Id,
                                    status = currentStatus.ToString(),
                                    paidUnits = invoice.PaidAmountUnits,
                                    expectedUnits = invoice.ExpectedAmountUnits,
                                    txHash = txHash
                                });

                                var outbox = new OutboxEvent(
                                    invoice.Id,
                                    $"invoice.{currentStatus.ToString().ToLowerInvariant()}",
                                    payload,
                                    invoice.CallbackUrl,
                                    merchant.EncryptedWebhookSecret);

                                _dbContext.OutboxEvents.Add(outbox);
                            }
                        }
                    }
                }

                cursor.UpdateCursor(blockTimestamp, txDetail.BlockNumber);
            }

            // Check next page fingerprint
            if (doc.RootElement.TryGetProperty("meta", out var meta) &&
                meta.TryGetProperty("fingerprint", out var fpProp) &&
                !string.IsNullOrWhiteSpace(fpProp.GetString()))
            {
                fingerprint = fpProp.GetString();
                cursor.SetFingerprint(fingerprint);
            }
            else
            {
                hasMorePages = false;
                cursor.SetFingerprint(null);
            }
        }
    }

    private async Task<ulong> GetSolidifiedBlockNumberAsync(CancellationToken ct)
    {
        try
        {
            var res = await _httpClient.GetAsync("https://api.trongrid.io/walletsolidity/getnowblock", ct);
            if (!res.IsSuccessStatusCode) return 0;

            var json = await res.Content.ReadAsStringAsync(ct);
            using var doc = JsonDocument.Parse(json);
            if (doc.RootElement.TryGetProperty("block_header", out var header) &&
                header.TryGetProperty("raw_data", out var raw) &&
                raw.TryGetProperty("number", out var num))
            {
                return (ulong)num.GetInt64();
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to fetch solidified head block from TronGrid.");
        }
        return 0;
    }

    private async Task<(bool IsSolidified, bool IsSuccess, ulong BlockNumber)> VerifySolidifiedTransactionAsync(string txHash, CancellationToken ct)
    {
        try
        {
            using var content = new StringContent(JsonSerializer.Serialize(new { value = txHash }), System.Text.Encoding.UTF8, "application/json");
            var res = await _httpClient.PostAsync("https://api.trongrid.io/walletsolidity/gettransactioninfobyid", content, ct);

            if (!res.IsSuccessStatusCode) return (false, false, 0);

            var json = await res.Content.ReadAsStringAsync(ct);
            using var doc = JsonDocument.Parse(json);

            if (doc.RootElement.ValueKind != JsonValueKind.Object || !doc.RootElement.TryGetProperty("blockNumber", out var bNum))
            {
                return (false, false, 0);
            }

            ulong blockNumber = (ulong)bNum.GetInt64();
            bool isSuccess = true;

            if (doc.RootElement.TryGetProperty("receipt", out var receipt) &&
                receipt.TryGetProperty("result", out var resultStr))
            {
                isSuccess = string.Equals(resultStr.GetString(), "SUCCESS", StringComparison.OrdinalIgnoreCase);
            }

            return (true, isSuccess, blockNumber);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error verifying solidified transaction for hash {TxHash}", txHash);
            return (false, false, 0);
        }
    }

using System;
using System.Security.Cryptography;
using System.Text;
using Gateway.Application.Contracts;
using Microsoft.Extensions.Configuration;

namespace Gateway.Infrastructure.Security;

public class AesGcmSecretProtector : ISecretProtector
{
    private readonly byte[] _masterKey;

    public AesGcmSecretProtector(IConfiguration configuration)
    {
        var keyHex = configuration["Security:MasterEncryptionKey"] 
                     ?? "0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF0123456789ABCDEF"; // 256-bit fallback for dev
        _masterKey = Convert.FromHexString(keyHex);
    }

    public string Encrypt(string plainText)
    {
        var plainBytes = Encoding.UTF8.GetBytes(plainText);
        var nonce = new byte[AesGcm.NonceByteSizes.MaxSize];
        RandomNumberGenerator.Fill(nonce);

        var cipherBytes = new byte[plainBytes.Length];
        var tag = new byte[AesGcm.TagByteSizes.MaxSize];

        using var aesGcm = new AesGcm(_masterKey, AesGcm.TagByteSizes.MaxSize);
        aesGcm.Encrypt(nonce, plainBytes, cipherBytes, tag);

        var combined = new byte[nonce.Length + tag.Length + cipherBytes.Length];
        Buffer.BlockCopy(nonce, 0, combined, 0, nonce.Length);
        Buffer.BlockCopy(tag, 0, combined, nonce.Length, tag.Length);
        Buffer.BlockCopy(cipherBytes, 0, combined, nonce.Length + tag.Length, cipherBytes.Length);

        return Convert.ToBase64String(combined);
    }

    public string Decrypt(string cipherText)
    {
        var combined = Convert.FromBase64String(cipherText);
        var nonceSize = AesGcm.NonceByteSizes.MaxSize;
        var tagSize = AesGcm.TagByteSizes.MaxSize;

        var nonce = new byte[nonceSize];
        var tag = new byte[tagSize];
        var cipherBytes = new byte[combined.Length - nonceSize - tagSize];

        Buffer.BlockCopy(combined, 0, nonce, 0, nonceSize);
        Buffer.BlockCopy(combined, nonceSize, tag, 0, tagSize);
        Buffer.BlockCopy(combined, nonceSize + tagSize, cipherBytes, 0, cipherBytes.Length);

        var plainBytes = new byte[cipherBytes.Length];
        using var aesGcm = new AesGcm(_masterKey, tagSize);
        aesGcm.Decrypt(nonce, cipherBytes, tag, plainBytes);

        return Encoding.UTF8.GetString(plainBytes);
    }
}

using System;
using System.Net;
using System.Net.Sockets;

namespace Gateway.Infrastructure.Security;

public static class SsrfValidator
{
    public static bool IsDestinationSafe(string urlString)
    {
        if (!Uri.TryCreate(urlString, UriKind.Absolute, out var uri))
            return false;

        // Force HTTPS in production environments
        if (uri.Scheme != Uri.UriSchemeHttps && uri.Scheme != Uri.UriSchemeHttp)
            return false;

        if (uri.IsLoopback || uri.HostNameType == UriHostNameType.IPv4 && IPAddress.IsLoopback(IPAddress.Parse(uri.Host)))
            return false;

        try
        {
            var ips = Dns.GetHostAddresses(uri.DnsSafeHost);
            if (ips.Length == 0) return false;

            foreach (var ip in ips)
            {
                if (IsProhibitedIp(ip))
                    return false;
            }
            return true;
        }
        catch
        {
            return false;
        }
    }

    public static bool IsProhibitedIp(IPAddress ip)
    {
        if (IPAddress.IsLoopback(ip)) return true;

        if (ip.AddressFamily == AddressFamily.InterNetwork)
        {
            var b = ip.GetAddressBytes();
            if (b[0] == 10) return true; // 10.0.0.0/8
            if (b[0] == 172 && b[1] >= 16 && b[1] <= 31) return true; // 172.16.0.0/12
            if (b[0] == 192 && b[1] == 168) return true; // 192.168.0.0/16
            if (b[0] == 169 && b[1] == 254) return true; // 169.254.0.0/16
            if (b[0] == 127) return true;
            if (b[0] == 0) return true;
        }
        else if (ip.AddressFamily == AddressFamily.InterNetworkV6)
        {
            if (ip.IsIPv6LinkLocal || ip.IsIPv6SiteLocal || ip.IsIPv6Multicast) return true;
            if (ip.Equals(IPAddress.IPv6Loopback) || ip.Equals(IPAddress.IPv6None)) return true;
        }

        return false;
    }
}

using System;
using System.Linq;
using System.Net.Http;
using System.Security.Cryptography;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using Gateway.Application.Contracts;
using Gateway.Infrastructure.Security;
using Gateway.Persistence;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace Gateway.Infrastructure.Webhooks;

public class WebhookDispatcher : IWebhookDispatcher
{
    private readonly GatewayDbContext _dbContext;
    private readonly HttpClient _httpClient;
    private readonly ILogger<WebhookDispatcher> _logger;

    public WebhookDispatcher(
        GatewayDbContext dbContext,
        HttpClient httpClient,
        ILogger<WebhookDispatcher> logger)
    {
        _dbContext = dbContext;
        _httpClient = httpClient;
        _logger = logger;
    }

    public async Task DispatchPendingWebhooksAsync(CancellationToken cancellationToken)
    {
        var now = DateTime.UtcNow;

        var pendingEvents = await _dbContext.OutboxEvents
            .Where(o => !o.IsProcessed && o.NextAttemptAtUtc != null && o.NextAttemptAtUtc <= now)
            .OrderBy(o => o.CreatedAtUtc)
            .Take(10)
            .ToListAsync(cancellationToken);

        if (!pendingEvents.Any())
            return;

        foreach (var ev in pendingEvents)
        {
            // 1. Enforce strict SSRF protection before initiating HTTP socket
            if (!SsrfValidator.IsDestinationSafe(ev.DestinationUrl))
            {
                _logger.LogWarning("Webhook destination {Url} blocked by SSRF filter for event {EventId}", ev.DestinationUrl, ev.Id);
                ev.RecordFailure("Destination blocked by SSRF validation rules.", maxRetries: 1);
                continue;
            }

            try
            {
                var timestamp = DateTimeOffset.UtcNow.ToUnixTimeSeconds().ToString();
                var signature = ComputeHmacSha256($"{timestamp}.{ev.Payload}", ev.SecretKey);

                using var request = new HttpRequestMessage(HttpMethod.Post, ev.DestinationUrl)
                {
                    Content = new StringContent(ev.Payload, Encoding.UTF8, "application/json")
                };

                // Add required security webhook headers
                request.Headers.Add("X-CryptoGateway-Event-Id", ev.Id.ToString());
                request.Headers.Add("X-CryptoGateway-Timestamp", timestamp);
                request.Headers.Add("X-CryptoGateway-Signature", signature);

                using var response = await _httpClient.SendAsync(request, cancellationToken);

                if (response.IsSuccessStatusCode)
                {
                    ev.RecordSuccess();
                    _logger.LogInformation("Webhook event {EventId} successfully delivered to {Url}", ev.Id, ev.DestinationUrl);
                }
                else
                {
                    var errorMsg = $"HTTP {(int)response.StatusCode}: {response.ReasonPhrase}";
                    ev.RecordFailure(errorMsg);
                    _logger.LogWarning("Webhook delivery failed for event {EventId}: {Status}", ev.Id, response.StatusCode);
                }
            }
            catch (Exception ex)
            {
                ev.RecordFailure(ex.Message);
                _logger.LogError(ex, "Exception while delivering webhook event {EventId}", ev.Id);
            }
        }

        await _dbContext.SaveChangesAsync(cancellationToken);
    }

    private static string ComputeHmacSha256(string data, string secret)
    {
        using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(secret));
        var hash = hmac.ComputeHash(Encoding.UTF8.GetBytes(data));
        return Convert.ToHexString(hash).ToLowerInvariant();
    }

using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;
using System.Threading.Tasks;
using Gateway.Application.Contracts;
using Gateway.Domain.Entities;
using Gateway.Persistence;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;

namespace Gateway.Api.Authentication;

[AttributeUsage(AttributeTargets.Class | AttributeTargets.Method)]
public class RequireMerchantSignatureAttribute : Attribute, IAsyncActionFilter
{
    public async Task OnActionExecutionAsync(ActionExecutingContext context, ActionExecutionDelegate next)
    {
        var request = context.HttpContext.Request;

        if (!request.Headers.TryGetValue("X-API-Key", out var apiKeyVals) ||
            !request.Headers.TryGetValue("X-Signature", out var sigVals) ||
            !request.Headers.TryGetValue("X-Timestamp", out var tsVals) ||
            !request.Headers.TryGetValue("X-Nonce", out var nonceVals))
        {
            context.Result = new UnauthorizedObjectResult(new { error = "Missing required authentication headers." });
            return;
        }

        var apiKey = apiKeyVals.ToString();
        var signature = sigVals.ToString();
        var nonce = nonceVals.ToString();

        if (!long.TryParse(tsVals.ToString(), out var timestampEpoch))
        {
            context.Result = new UnauthorizedObjectResult(new { error = "Invalid timestamp format." });
            return;
        }

        var reqTime = DateTimeOffset.FromUnixTimeSeconds(timestampEpoch).UtcDateTime;
        if (Math.Abs((DateTime.UtcNow - reqTime).TotalMinutes) > 5)
        {
            context.Result = new UnauthorizedObjectResult(new { error = "Request timestamp expired or skewed beyond 5 minutes." });
            return;
        }

        var db = context.HttpContext.RequestServices.GetRequiredService<GatewayDbContext>();
        var protector = context.HttpContext.RequestServices.GetRequiredService<ISecretProtector>();

        using var sha = SHA256.Create();
        var keyHash = Convert.ToHexString(sha.ComputeHash(Encoding.UTF8.GetBytes(apiKey))).ToLowerInvariant();
        var merchant = await db.Merchants.FirstOrDefaultAsync(m => m.ApiKeyHash == keyHash && m.IsActive);

        if (merchant == null)
        {
            context.Result = new UnauthorizedObjectResult(new { error = "Invalid merchant credentials." });
            return;
        }

        // Atomic replay prevention check
        var existingNonce = await db.MerchantNonces
            .AnyAsync(n => n.MerchantId == merchant.Id && n.Nonce == nonce);

        if (existingNonce)
        {
            context.Result = new ConflictObjectResult(new { error = "Replay detected: Nonce has already been used." });
            return;
        }

        db.MerchantNonces.Add(new MerchantNonce(merchant.Id, nonce, TimeSpan.FromMinutes(10)));
        try
        {
            await db.SaveChangesAsync();
        }
        catch (DbUpdateException)
        {
            context.Result = new ConflictObjectResult(new { error = "Replay detected: Duplicate concurrent nonce." });
            return;
        }

        // Canonical Request Verification
        request.EnableBuffering();
        string rawBody;
        using (var reader = new StreamReader(request.Body, Encoding.UTF8, leaveOpen: true))
        {
            rawBody = await reader.ReadToEndAsync();
            request.Body.Position = 0;
        }

        var bodyHash = Convert.ToHexString(sha.ComputeHash(Encoding.UTF8.GetBytes(rawBody))).ToLowerInvariant();
        var canonical = $"{request.Method.ToUpperInvariant()}\n{request.Path}\n{timestampEpoch}\n{nonce}\n{bodyHash}";

        var secret = protector.Decrypt(merchant.EncryptedApiKeySecret);
        using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(secret));
        var expectedSig = Convert.ToHexString(hmac.ComputeHash(Encoding.UTF8.GetBytes(canonical))).ToLowerInvariant();

        if (!CryptographicOperations.FixedTimeEquals(
            Encoding.UTF8.GetBytes(signature.ToLowerInvariant()),
            Encoding.UTF8.GetBytes(expectedSig)))
        {
            context.Result = new UnauthorizedObjectResult(new { error = "Invalid HMAC signature." });
            return;
        }

        context.HttpContext.Items["MerchantId"] = merchant.Id;
        await next();
    }
}

using System;

[ApiController]
[Route("api/v1/checkout")]
public class CheckoutController : ControllerBase
{
    private readonly GatewayDbContext _dbContext;

    public CheckoutController(GatewayDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    [HttpGet("{token}")]
    public async Task<IActionResult> GetCheckoutStatus(string token, CancellationToken ct)
    {
        if (string.IsNullOrWhiteSpace(token))
            return BadRequest(new { error = "Checkout token is required." });

        var invoice = await _dbContext.Invoices
            .Include(i => i.Transfers)
            .FirstOrDefaultAsync(i => i.CheckoutToken == token, ct);

        if (invoice == null)
            return NotFound(new { error = "Payment checkout session not found." });

        // Update expiry in memory for display
        invoice.EvaluateStatus();

        var remainingSeconds = Math.Max(0, (long)(invoice.ExpiresAtUtc - DateTime.UtcNow).TotalSeconds);

        return Ok(new CheckoutStatusDto
        {
            DepositAddress = invoice.DepositAddress,
            ExpectedAmount = invoice.ExpectedAmountUnits / 1_000_000m,
            PaidAmount = invoice.PaidAmountUnits / 1_000_000m,
            Currency = invoice.Currency,
            Network = invoice.Network,
            Status = invoice.Status.ToString(),
            RemainingSeconds = remainingSeconds
        });
    }
}

using System;

namespace Gateway.Api.Controllers;

[ApiController]
[Route("api/v1/invoices")]
public class InvoicesController : ControllerBase
{
    private readonly GatewayDbContext _dbContext;

    public InvoicesController(GatewayDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    [HttpPost]
    [RequireMerchantSignature]
    public async Task<IActionResult> CreateInvoice([FromBody] CreateInvoiceRequest request, CancellationToken ct)
    {
        var merchantId = (Guid)HttpContext.Items["MerchantId"]!;

        // Restrict fiat currency support explicitly (Phase E requirement)
        var fiatCurrency = request.FiatCurrency.Trim().ToUpperInvariant();
        if (fiatCurrency != "USD")
        {
            return BadRequest(new { error = $"Unsupported fiat currency '{fiatCurrency}'. Only 'USD' is supported currently." });
        }

        await using var tx = await _dbContext.Database.BeginTransactionAsync(IsolationLevel.ReadCommitted, ct);

        // Exclusive row lock on the wallet row to prevent concurrent derivation collisions
        var wallet = await _dbContext.Wallets
            .FromSqlInterpolated($"SELECT * FROM \"Wallets\" WHERE \"MerchantId\" = {merchantId} AND \"Network\" = 'TRON' FOR UPDATE")
            .FirstOrDefaultAsync(ct);

        if (wallet == null)
        {
            return BadRequest(new { error = "Merchant TRON wallet is not configured." });
        }

        var derivationService = new TronAddressDerivationService(wallet.AccountXPub);
        var derivedAddressStr = derivationService.DeriveAddress(wallet.LastDerivationIndex);
        var derivedAddressEntity = wallet.AllocateNextAddress(derivedAddressStr);

        // Explicit 1.0 USD -> 1.0 USDT snapshot pegged rate
        const decimal exchangeRate = 1.0m;
        var cryptoAmount = request.FiatAmount / exchangeRate;
        var expectedUnits = (long)Math.Round(cryptoAmount * 1_000_000m, MidpointRounding.AwayFromZero);
        var underpaymentThreshold = 50_000L;

        var checkoutToken = Convert.ToHexString(RandomNumberGenerator.GetBytes(32)).ToLowerInvariant();

        var invoice = new Invoice(
            merchantId: merchantId,
            derivedAddressId: derivedAddressEntity.Id,
            depositAddress: derivedAddressStr,
            fiatAmount: request.FiatAmount,
            fiatCurrency: fiatCurrency,
            exchangeRate: exchangeRate,
            exchangeRateSource: "DirectPegProvider",
            exchangeRateTimestampUtc: DateTime.UtcNow,
            expectedAmountUnits: expectedUnits,
            underpaymentThresholdUnits: underpaymentThreshold,
            callbackUrl: request.CallbackUrl,
            checkoutToken: checkoutToken,
            validityDuration: TimeSpan.FromMinutes(request.ExpiryMinutes)
        );

        _dbContext.Invoices.Add(invoice);
        await _dbContext.SaveChangesAsync(ct);
        await tx.CommitAsync(ct);

        var response = new InvoiceResponseDto
        {
            InvoiceId = invoice.Id,
            DepositAddress = invoice.DepositAddress,
            FiatAmount = invoice.FiatAmount,
            FiatCurrency = invoice.FiatCurrency,
            ExchangeRate = invoice.ExchangeRate,
            ExpectedCryptoAmount = cryptoAmount,
            ExpectedUnits = invoice.ExpectedAmountUnits,
            Currency = invoice.Currency,
            Network = invoice.Network,
            Status = invoice.Status.ToString(),
            CheckoutUrl = $"{Request.Scheme}://{Request.Host}/checkout.html?token={invoice.CheckoutToken}",
            ExpiresAtUtc = invoice.ExpiresAtUtc,
            CreatedAtUtc = invoice.CreatedAtUtc
        };

        return CreatedAtAction(nameof(GetInvoiceById), new { invoiceId = invoice.Id }, response);
    }

    [HttpGet("{invoiceId:guid}")]
    [RequireMerchantSignature]
    public async Task<IActionResult> GetInvoiceById(Guid invoiceId, CancellationToken ct)
    {
        var merchantId = (Guid)HttpContext.Items["MerchantId"]!;

        var invoice = await _dbContext.Invoices
            .FirstOrDefaultAsync(i => i.Id == invoiceId && i.MerchantId == merchantId, ct);

            DepositAddress = invoice.DepositAddress,
            FiatAmount = invoice.FiatAmount,
            FiatCurrency = invoice.FiatCurrency,
            ExchangeRate = invoice.ExchangeRate,
            ExpectedCryptoAmount = invoice.ExpectedAmountUnits / 1_000_000m,
            ExpectedUnits = invoice.ExpectedAmountUnits,

            CheckoutUrl = $"{Request.Scheme}://{Request.Host}/checkout.html?token={invoice.CheckoutToken}",

using System;
using System.Threading;
using System.Threading.Tasks;
using Gateway.Application.Contracts;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

namespace Gateway.Worker;

public class Worker : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<Worker> _logger;

    public Worker(IServiceProvider serviceProvider, ILogger<Worker> logger)
    {
        _serviceProvider = serviceProvider;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Tron Scanner Background Worker started.");

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = _serviceProvider.CreateScope();
                var scanner = scope.ServiceProvider.GetRequiredService<ITronScannerService>();
                await scanner.ScanPendingDepositsAsync(stoppingToken);

                // Dispatch Transactional Outbox Webhook events
                var webhookDispatcher = scope.ServiceProvider.GetRequiredService<IWebhookDispatcher>();
                await webhookDispatcher.DispatchPendingWebhooksAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Unhandled error in worker execution loop.");
            }

            // Standard TRON block time is 3 seconds; poll every 5 seconds
            await Task.Delay(TimeSpan.FromSeconds(5), stoppingToken);
        }
    }
}
